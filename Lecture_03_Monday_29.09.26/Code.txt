# match only those lines that are not empty, and are not the header, and count the number of lines

grep -v "^$" example_people_data.tsv | grep -v "name" > "example_people_data2.tsv"
# save the list of people (without empty lines and headers) to another file

# 29/09/1996
awk 'BEGIN{FS="\t";} ($4<=29) && ($5<=9) && ($6<=1996)' example_people_data2.tsv | wc -l

cut -f1 example_people_data2.tsv | grep -i "Jan$" | wc -l

cut -f7 example_people_data2.tsv | sort | uniq -c | awk 'BEGIN{FS=" "} {if($1>counter) {counter=$1; country=$2}} END {print country}'
# extract column 7 (list of countries); sort them and count number of unique countries
# 'counter' is an uninitialised variable so by default has value 0
# loop through each line (with count of countries as $1) and return the country with the highest value

awk 'BEGIN{FS="\t"} ($7=="Mozambique") && ($4<=29) && ($5<=9) && ($6<=1976)' example_people_data2.tsv | wc -l
# (would be nice to combine 4 + 5)

sort -r -k7,7 example_people_data2.tsv | awk 'BEGIN{FS="\t"} {print $1, "\t", $2}' | grep ".edu$" | awk 'BEGIN{FS="\t"} {print $1} > names.txt
# sort rows by reverse alphabetical order of countries
# extract columns 1 and 2 (names and email)
# return rows that end in .edu
